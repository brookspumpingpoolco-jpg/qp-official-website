    <script>
        const RENTAL_SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbxfSwpb2EKeibKH0fyolMgAAfNPz1w18SYjVQqc63DPdWJWIgTLDp2iHI4z_ilp8I8Tsg/exec';
        const DASHBOARD_URL = 'https://sites.google.com/view/mcafeepropertiesllc/rental-management-dashboard';
        const loginForm = document.getElementById('loginForm');
        const loginButton = document.getElementById('loginButton');
        const loadingIcon = document.getElementById('loadingIcon');
        const buttonText = document.getElementById('buttonText');
        const errorMessage = document.getElementById('errorMessage');
        function showError(m) { errorMessage.textContent = m; errorMessage.classList.add('show'); setTimeout(() => errorMessage.classList.remove('show'), 5000); }
        function setLoading(l) { if (l) { loadingIcon.classList.add('show'); buttonText.textContent = 'Signing in...'; loginButton.disabled = true; } else { loadingIcon.classList.remove('show'); buttonText.textContent = 'Sign In'; loginButton.disabled = false; } }
        loginForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            setLoading(true);
            errorMessage.classList.remove('show');
            const email = document.getElementById('email').value.trim();
            const password = document.getElementById('password').value;
            const keepSignedIn = document.getElementById('keepSignedIn').checked;
            try {
                const formData = new URLSearchParams();
                formData.append('action', 'login');
                formData.append('email', email);
                formData.append('password', password);
                const response = await fetch(RENTAL_SCRIPT_URL, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: formData.toString() });
                const data = await response.json();
                if (data.success) {
                    if (keepSignedIn) { localStorage.setItem('rental_userEmail', email); localStorage.setItem('rental_userName', data.user.name); localStorage.setItem('rental_userRole', data.user.role); } else { sessionStorage.setItem('rental_userEmail', email); sessionStorage.setItem('rental_userName', data.user.name); sessionStorage.setItem('rental_userRole', data.user.role); }
                    window.location.href = DASHBOARD_URL;
                } else { showError(data.message || 'Login failed'); setLoading(false); }
            } catch (error) { showError('Connection error'); setLoading(false); }
        });
        const storedEmail = localStorage.getItem('rental_userEmail') || sessionStorage.getItem('rental_userEmail');
        if (storedEmail) { window.location.href = DASHBOARD_URL; }
    </script>
